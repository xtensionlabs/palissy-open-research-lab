import numpy as np, mathnp.random.seed(42)                      # deterministic N_GENES = 200rcbs = np.random.beta(2, 5, N_GENES)    # codon bias strength per genetrna = np.random.normal(0, 0.15, N_GENES)expr = 0.8 * rcbs + trna                # expression = bias + tRNA supply noise def ranks(a):    return np.argsort(np.argsort(a)).astype(float) def spearman(x, y):    rx, ry = ranks(x), ranks(y)    return np.corrcoef(rx, ry)[0, 1] rho = spearman(rcbs, expr)perm = [abs(spearman(rcbs, np.random.permutation(expr))) for _ in range(999)]p = (sum(v >= abs(rho) for v in perm) + 1) / 1000 print(f"Observed Spearman rho: {rho:.4f}")print(f"Empirical p-value (two-sided): {p:.4f}")print("RESULT: supports" if rho > 0.5 and p < 0.01 else "RESULT: inconclusive")
